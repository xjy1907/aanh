bool isShowMenu = true;

@implementation ImGuiDrawView

- (instancetype)initWithFrame:(CGRect)frame
{
    if (self = [super initWithFrame:frame]) {
        self.backgroundColor = [UIColor clearColor];
        self.userInteractionEnabled = NO;
        [self setupUI];
    }
    
    screenSize = [UIScreen mainScreen].bounds.size;
    screenSize.width *= [UIScreen mainScreen].nativeScale;
    screenSize.height *= [UIScreen mainScreen].nativeScale;
    
    return self;
    
}

#pragma mark - UI初始化

- (void)setupUI
{
    self.mtkView = [[MTKView alloc] initWithFrame:[UIScreen mainScreen].bounds];
    self.mtkView.backgroundColor = [UIColor clearColor];
    [self addSubview:self.mtkView];
    
    self.mtkView.device = MTLCreateSystemDefaultDevice();
    if (!self.mtkView.device) return;
    self.mtkView.preferredFramesPerSecond = 120;
    self.renderer = [[ImGuiMTKView alloc] initWithView:self.mtkView];
    self.renderer.delegate = self;
    self.mtkView.delegate = self.renderer;
    [self.renderer initializePlatform];
}

bool LoadingCircle(ImVec2 vecCenter)
{
    static ImColor fore_color(200, 20, 20, 255);
    static ImColor back_color(200, 20, 20, 40);
    
    auto front = ImGui::GetForegroundDrawList();
    float time = ImGui::GetTime();
    
    alpha = min_alpha + (max_alpha - min_alpha) * (sinf(2 * IM_PI * time / pulse_duration) + 1) / 2;
    
    position += ImGui::GetIO().DeltaTime * (8.f + rotationCount);
    
    if (position >= IM_PI * 2.f) { // »ÁÏÂÌË?ÛÒÎÓ‚Ë?Ò·ÓÒ‡
        position -= IM_PI * 2.f;
        rotationCount++;            // ”‚ÂÎË˜Ë‚‡Â?Ò˜ÂÚ˜Ë?Ó·ÓÓÚÓ‚
    }
    
    // ≈ÒÎË ·˚ÎÓ ‰ÓÒÚË„ÌÛÚÓ 3 Ó·ÓÓÚ? Ì‡˜ËÌ‡ÂÏ ÛÏÂÌ¸¯‡Ú?‡Î¸Ù?Í‡Ì‡?
    if (rotationCount >= 8)
        alphaAnimation -= ImGui::GetIO().DeltaTime;
    
    // Õ‡˜ËÌ‡ÂÏ ÛÏÂÌ¸¯‡Ú?ÁÌ‡˜ÂÌËÂ ‡Î¸Ù?
    if (alphaAnimation < 0.f) {
        alphaAnimation = 0.f;
    }
    
    radius = ImLerp(radius, rotationCount <= 5 ? 75 : rotationCount > 5 && rotationCount < 8 ? 55.f : rotationCount >= 7 ? 85 : 0.f, anim_speed);
    thickness = ImLerp(thickness, rotationCount <= 8 ? 8.f : 0.f, anim_speed);
    
    alpha_circle = ImLerp(alpha_circle, rotationCount >= 5 && rotationCount < 8 ? 1.f : 0.f, anim_speed);
    alpha_icon[0] = ImLerp(alpha_icon[0], rotationCount < 5 ? 1.f : 0.f, anim_speed);
    alpha_icon[1] = ImLerp(alpha_icon[1], rotationCount >= 5 && rotationCount < 8 ? 1.f : 0.f, anim_speed);
    
    // Background
    front->PathClear();
    front->PathArcTo(vecCenter, radius, 0.f, 2.f * IM_PI, 40.f);
    front->PathStroke(GetColorWithAlpha(ImGui::GetColorU32(background_color), alphaAnimation / 3), false, thickness);
    
    front->PathClear();
    front->PathArcTo(vecCenter, radius, IM_PI * 1.5f + position, IM_PI * (1.5f + arc_size) + position, 40.f);
    front->PathStroke(GetColorWithAlpha(ImGui::GetColorU32(main_color), alpha * alphaAnimation), false, thickness); // œËÏÂÌˇÂÏ alphaAnimation
    
    front->AddCircleFilled(vecCenter, radius + thickness / 2, GetColorWithAlpha(ImGui::GetColorU32(main_color), alpha_circle));
    
    // ≈ÒÎË ‡Î¸Ù?‰ÓÒÚË„Î‡ 0 - ‚ÓÁ‚‡˘‡ÂÏ true.
    if (alphaAnimation <= 0.f)
        return true;
    
    return false;
}

namespace ImGui
{
    int rotation_start_index;
    
    void ImRotateStart()
    {
        rotation_start_index = ImGui::GetWindowDrawList()->VtxBuffer.Size;
    }
    
    ImVec2 ImRotationCenter()
    {
        ImVec2 l(FLT_MAX, FLT_MAX), u(-FLT_MAX, -FLT_MAX); // bounds
        
        const auto& buf = ImGui::GetWindowDrawList()->VtxBuffer;
        for (int i = rotation_start_index; i < buf.Size; i++)
        {
            l = ImMin(l, buf[i].pos);
            u = ImMax(u, buf[i].pos);
        }
        
        return ImVec2((l.x + u.x) / 2, (l.y + u.y) / 2); // or use _ClipRectStack?
    }
    
    inline ImVec2 operator-(const ImVec2& lhs, const ImVec2& rhs)
    {
        return ImVec2(lhs.x - rhs.x, lhs.y - rhs.y);
    }
    
    void ImRotateEnd(float rad, ImVec2 center = ImRotationCenter())
    {
        float s = sin(rad), c = cos(rad);
        ImVec2 rotated_center = ImRotate(center, s, c);
        center = rotated_center - center;
        
        auto& buf = ImGui::GetWindowDrawList()->VtxBuffer;
        for (int i = rotation_start_index; i < buf.Size; i++)
        {
            buf[i].pos = ImRotate(buf[i].pos, s, c) - center;
        }
    }
}

void Trinage_background()
{
    
    ImVec2 screen_size = ImVec2([UIScreen mainScreen].bounds.size.width * 2,[UIScreen mainScreen].bounds.size.height * 2);
    
    static ImVec2 partile_pos[100*2];
    static ImVec2 partile_target_pos[100*2];
    static float partile_speed[100];
    static float partile_size[100*2];
    static float partile_radius[100*2];
    static float partile_rotate[100*2];
    
    for (int i = 1; i < 50; i++)
    {
        if (partile_pos[i].x == 0 || partile_pos[i].y == 0)
        {
            partile_pos[i].x = rand() % (int)screen_size.x + 1*3;
            partile_pos[i].y = 15*3.f;
            partile_speed[i] = 1 + rand() % 25;
            partile_radius[i] = rand() % 4*2;
            partile_size[i] = rand() % 8*2;
            
            partile_target_pos[i].x = rand() % (int)screen_size.x;
            partile_target_pos[i].y = screen_size.y * 2*3;
        }
        
        partile_pos[i] = ImLerp(partile_pos[i], partile_target_pos[i], ImGui::GetIO().DeltaTime * (partile_speed[i] / 60));
        partile_rotate[i] += ImGui::GetIO().DeltaTime;
        
        if (partile_pos[i].y > screen_size.y)
        {
            partile_pos[i].x = 0;
            partile_pos[i].y = 0;
            partile_rotate[i] = 0;
        }
        
        ImGui::ImRotateStart();
        ImGui::GetWindowDrawList()->AddCircleFilled(partile_pos[i], partile_size[i], ImGui::GetColorU32(main_color), 2);
        ImGui::ImRotateEnd(partile_rotate[i]);
    }
}

uint64_t GetTickCount()
{
    return std::chrono::duration_cast<std::chrono::milliseconds>(
                                                                 std::chrono::steady_clock::now().time_since_epoch())
    .count();
}

namespace ImGui
{
    int rotation_start_index;
    
    void ImRotateStart()
    {
        rotation_start_index = ImGui::GetWindowDrawList()->VtxBuffer.Size;
    }
    
    ImVec2 ImRotationCenter()
    {
        ImVec2 l(FLT_MAX, FLT_MAX), u(-FLT_MAX, -FLT_MAX); // bounds
        
        const auto& buf = ImGui::GetWindowDrawList()->VtxBuffer;
        for (int i = rotation_start_index; i < buf.Size; i++)
        {
            l = ImMin(l, buf[i].pos);
            u = ImMax(u, buf[i].pos);
        }
        
        return ImVec2((l.x + u.x) / 2, (l.y + u.y) / 2); // or use _ClipRectStack?
    }
    
    inline ImVec2 operator-(const ImVec2& lhs, const ImVec2& rhs)
    {
        return ImVec2(lhs.x - rhs.x, lhs.y - rhs.y);
    }
    
    void ImRotateEnd(float rad, ImVec2 center = ImRotationCenter())
    {
        float s = sin(rad), c = cos(rad);
        ImVec2 rotated_center = ImRotate(center, s, c);
        center = rotated_center - center;
        
        auto& buf = ImGui::GetWindowDrawList()->VtxBuffer;
        for (int i = rotation_start_index; i < buf.Size; i++)
        {
            buf[i].pos = ImRotate(buf[i].pos, s, c) - center;
        }
    }
}

void Trinage_background()
{
    
    ImVec2 screen_size = ImVec2([UIScreen mainScreen].bounds.size.width * 2,[UIScreen mainScreen].bounds.size.height * 2);
    
    static ImVec2 partile_pos[100*2];
    static ImVec2 partile_target_pos[100*2];
    static float partile_speed[100];
    static float partile_size[100*2];
    static float partile_radius[100*2];
    static float partile_rotate[100*2];
    
    for (int i = 1; i < 50; i++)
    {
        if (partile_pos[i].x == 0 || partile_pos[i].y == 0)
        {
            partile_pos[i].x = rand() % (int)screen_size.x + 1*3;
            partile_pos[i].y = 15*3.f;
            partile_speed[i] = 1 + rand() % 25;
            partile_radius[i] = rand() % 4*2;
            partile_size[i] = rand() % 8*2;
            
            partile_target_pos[i].x = rand() % (int)screen_size.x;
            partile_target_pos[i].y = screen_size.y * 2*3;
        }
        
        partile_pos[i] = ImLerp(partile_pos[i], partile_target_pos[i], ImGui::GetIO().DeltaTime * (partile_speed[i] / 60));
        partile_rotate[i] += ImGui::GetIO().DeltaTime;
        
        if (partile_pos[i].y > screen_size.y)
        {
            partile_pos[i].x = 0;
            partile_pos[i].y = 0;
            partile_rotate[i] = 0;
        }
        
        ImGui::ImRotateStart();
        ImGui::GetWindowDrawList()->AddCircleFilled(partile_pos[i], partile_size[i], ImGui::GetColorU32(main_color), 2);
        ImGui::ImRotateEnd(partile_rotate[i]);
    }
}
